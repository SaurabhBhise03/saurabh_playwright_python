# Verify pip is running out of your active virtual environment
pip --version

# Install all framework requirements at once
pip install -r requirements.txt

# Manual installation of individual plugins and extensions if requirements.txt is absent
pip install playwright pytest-playwright pytest-xdist allure-pytest openpyxl pillow

# Download the isolated Chromium browser binaries into your active virtual environment
python -m playwright install chromium

# Verify pytest successfully registers all parallel xdist and reporting plugins
pytest --version

# Run the test matrix parallelly across 3 active, visible browser windows simultaneously
python -m pytest -v -n 3 --headed

# Execute tests visually with an added 500ms delay between actions to clearly track step behaviors
python -m pytest -v -n 3 --headed --slowmo 500

#runtime:
pip install -r requirements.txt
python -m playwright install chromium install-deps

#Execute tests via the pytest-xdist parallel worker allocation strategy:
python -m pytest -v -n 3



# Automatically tracks host CPU logical configurations allocating proportional isolated threads
pytest -n auto --dist=loadfile --alluredir=allure-results

# Enforces hard constraints capping thread pools to prevent resource exhaustion on low-spec CI nodes
pytest -n 4 --dist=loadgroup --alluredir=allure-results


@pytest.mark.ui: Placed above the TestSauceUiAutomation class to group all web UI features.
@pytest.mark.api: Placed above the TestReqResApiAutomation class to group backend REST requests.
@pytest.mark.auth: Placed specifically above test_generate_auth_state and test_consume_auth_state to pinpoint token manipulation tests.
@pytest.mark.integration: Placed above your hybrid API-to-UI flow.

# Execute only the web interface test cases (Headed, 3 parallel windows)
python -m pytest -v -m ui -n 3 --headed

# Execute only the fast backend REST API CRUD verification suites
python -m pytest -v -m api

# Execute only the token preservation and storage state test cases
python -m pytest -v -m auth --headed

# Execute only the hybrid cross-layer end-to-end integration workflows
python -m pytest -v -m integration --headed

# Complex filter: Run every single test scenario in the framework EXCEPT the API suites
python -m pytest -v -m "not api" -n 2 --headed
